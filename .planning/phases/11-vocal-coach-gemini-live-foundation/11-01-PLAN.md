```typescript
import { NextRequest, NextResponse } from "next/server";
import { requireAuth } from "@/lib/api-auth";
import { checkRateLimit } from "@/lib/rate-limit";

/**
 * POST /api/gemini/live-token
 *
 * Génère un ephemeral token Gemini Live pour le client WebSocket.
 * La clé maître GEMINI_API_KEY reste serveur — jamais exposée côté client.
 *
 * Auth: Supabase session OU X-Demo-Mode: true (bypass + rate-limit IP)
 * Rate-limit: 10 tokens/min/user (demo: 10/min/IP)
 */

const GEMINI_API_KEY = process.env.GEMINI_API_KEY;
const GEMINI_LIVE_MODEL =
  process.env.GEMINI_LIVE_MODEL ?? "gemini-2.0-flash-live-001";

export async function POST(request: NextRequest): Promise<NextResponse> {
  // 1. Demo mode bypass (même pattern que /api/voice/tts)
  const isDemoMode = request.headers.get("X-Demo-Mode") === "true";

  let rateLimitKey: string;
  if (isDemoMode) {
    const fwd = request.headers.get("x-forwarded-for") ?? "unknown";
    rateLimitKey = `gemini-live-token-demo:${fwd}`;
  } else {
    const auth = await requireAuth();
    if (auth.error) return auth.error;
    rateLimitKey = `gemini-live-token:${auth.user.id}`;
  }

  // 2. Rate-limit : 10 tokens/min/user
  const { allowed } = checkRateLimit(rateLimitKey, 10, 60_000);
  if (!allowed) {
    return NextResponse.json({ error: "Trop de requêtes" }, { status: 429 });
  }

  // 3. Demo mode : retourner un token stubbé
  if (isDemoMode) {
    return NextResponse.json({
      token: `demo-live-token-${Date.now()}`,
      expiresAt: new Date(Date.now() + 5 * 60 * 1000).toISOString(),
      demo: true,
    });
  }

  // 4. Vérifier la clé API
  if (!GEMINI_API_KEY) {
    return NextResponse.json(
      { error: "GEMINI_API_KEY non configuré" },
      { status: 500 }
    );
  }

  // 5. Générer l'ephemeral token via @google/genai
  try {
    // Tentative via SDK @google/genai v1.48.0
    // Si GoogleGenAI.live.createEphemeralToken n'est pas disponible, fallback REST
    const { GoogleGenAI } = await import("@google/genai");
    const ai = new GoogleGenAI({ apiKey: GEMINI_API_KEY });

    // Note : l'API ephemeral tokens est en v1beta — vérifier disponibilité SDK
    // Si ai.live?.createEphemeralToken n'existe pas, utiliser REST directement
    if (typeof (ai as unknown as Record<string, unknown>).live === "object") {
      const liveClient = (ai as unknown as {
        live: {
          createEphemeralToken: (opts: { model: string }) => Promise<{ token: string; expiresAt: string }>;
        };
      }).live;
      const result = await liveClient.createEphemeralToken({ model: GEMINI_LIVE_MODEL });
      return NextResponse.json({ token: result.token, expiresAt: result.expiresAt });
    }

    // Fallback REST direct (v1beta)
    const resp = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/ephemeralTokens`,
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${GEMINI_API_KEY}`,
        },
        body: JSON.stringify({ model: GEMINI_LIVE_MODEL, ttl: "300s" }),
      }
    );

    if (!resp.ok) {
      const err = await resp.text();
      console.error("[gemini/live-token] API error:", resp.status, err);
      return NextResponse.json(
        { error: "Impossible de générer le token Gemini Live" },
        { status: 502 }
      );
    }

    const data = (await resp.json()) as { name?: string; token?: string; expireTime?: string };
    const token = data.token ?? data.name ?? "";
    const expiresAt = data.expireTime ?? new Date(Date.now() + 5 * 60 * 1000).toISOString();

    return NextResponse.json({ token, expiresAt });
  } catch (err) {
    console.error("[gemini/live-token] Erreur:", err instanceof Error ? err.message : err);
    return NextResponse.json(
      { error: "Erreur interne lors de la génération du token" },
      { status: 500 }
    );
  }
}
```

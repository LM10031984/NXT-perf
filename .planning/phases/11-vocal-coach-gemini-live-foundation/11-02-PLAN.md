Le worklet reçoit du navigateur un signal à la fréquence native (souvent 48 kHz) et doit produire des frames PCM 16 kHz mono que Gemini Live attend.

```javascript
/**
 * pcm-processor.js — AudioWorklet pour Gemini Live
 * Resampling mono 16 kHz — format attendu par Gemini Live (audio/pcm;rate=16000)
 * Chargé via : audioContext.audioWorklet.addModule('/audio-worklets/pcm-processor.js')
 *
 * Technique : accumulation de frames de 128 samples (taille native WebAudio),
 * envoi par blocs de 1024 samples pour limiter le trafic postMessage.
 */
const TARGET_SAMPLE_RATE = 16000;
const SEND_BLOCK_SAMPLES = 1024; // ~64ms à 16 kHz

class PcmProcessor extends AudioWorkletProcessor {
  constructor() {
    super();
    this._buffer = new Float32Array(0);
    // sampleRate est disponible globalement dans le contexte AudioWorklet
  }

  /**
   * Downsample simplifié : prendre 1 sample sur N (nearest-neighbour).
   * Suffisant pour la voix ; évite dépendance externe.
   */
  _downsample(input, inputRate) {
    if (inputRate === TARGET_SAMPLE_RATE) return input;
    const ratio = inputRate / TARGET_SAMPLE_RATE;
    const outputLength = Math.floor(input.length / ratio);
    const output = new Float32Array(outputLength);
    for (let i = 0; i < outputLength; i++) {
      output[i] = input[Math.floor(i * ratio)];
    }
    return output;
  }

  process(inputs, _outputs, _params) {
    const input = inputs[0];
    if (!input || !input[0]) return true;

    // Prendre uniquement le canal 0 (mono)
    const channel = input[0];

    // Resampling vers 16 kHz
    const resampled = this._downsample(channel, sampleRate);

    // Accumuler dans le buffer
    const combined = new Float32Array(this._buffer.length + resampled.length);
    combined.set(this._buffer);
    combined.set(resampled, this._buffer.length);
    this._buffer = combined;

    // Envoyer par blocs de SEND_BLOCK_SAMPLES
    while (this._buffer.length >= SEND_BLOCK_SAMPLES) {
      const block = this._buffer.slice(0, SEND_BLOCK_SAMPLES);
      this._buffer = this._buffer.slice(SEND_BLOCK_SAMPLES);
      this.port.postMessage({ type: "audio", data: block });
    }

    return true; // keep processor alive
  }
}

registerProcessor("pcm-processor", PcmProcessor);
```

Ce fichier est en JavaScript pur (pas TypeScript) car les AudioWorklets sont chargés comme assets statiques hors du pipeline tsc.

**A. Mettre à jour `coaching-method.ts`** pour exporter une fonction de lecture avec token budget check (per D-03 de 08-CONTEXT.md) :

```typescript
import * as fs from "fs";
import * as path from "path";

/**
 * Token budget soft cap pour la méthode dans le prompt (D-03).
 * ~4 chars/token en FR = 2000 tokens → 8000 chars.
 */
const TOKEN_BUDGET_CHARS = 8_000;

/**
 * Lit coaching-method.md depuis disk et retire le frontmatter YAML.
 * Utilisé uniquement côté serveur (Node runtime — route.ts).
 * @returns contenu Markdown sans frontmatter, cappé à TOKEN_BUDGET_CHARS
 */
export function readCoachingMethod(): string {
  const filePath = path.join(process.cwd(), "src/data/coaching-method/coaching-method.md");
  let raw: string;
  try {
    raw = fs.readFileSync(filePath, "utf-8");
  } catch {
    // Fallback gracieux : utiliser la constante TypeScript si le fichier est absent
    console.warn("[coaching-method] coaching-method.md introuvable, fallback sur constante TS");
    return COACHING_METHOD_NXT;
  }
  // Retirer le frontmatter YAML (--- ... ---)
  const withoutFrontmatter = raw.replace(/^---[\s\S]*?---\s*/m, "");
  // Token budget : si > TOKEN_BUDGET_CHARS, tronquer à la dernière ligne complète
  if (withoutFrontmatter.length <= TOKEN_BUDGET_CHARS) return withoutFrontmatter.trim();
  const truncated = withoutFrontmatter.slice(0, TOKEN_BUDGET_CHARS);
  const lastNewline = truncated.lastIndexOf("\n");
  return (lastNewline > 0 ? truncated.slice(0, lastNewline) : truncated).trim();
}

// Constante conservée pour compatibilité arrière et tests offline
export const COACHING_METHOD_NXT = `# Méthode coaching NXT
[... contenu existant inchangé ...]
`;
```

Conserver `COACHING_METHOD_NXT` existante intacte — uniquement ajouter `readCoachingMethod()`.
